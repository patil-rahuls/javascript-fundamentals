## Nullish Coalescing Assignment Operator (**`??=`**)

> 🎯 This operator evaluates *and* assigns a value. It assigns the right operand to the left operand **only if** the left operand is nullish (**`null`** or **`undefined`**).
> 
> It prevents overwriting valid falsy values like **`0`**, **`false`**, or **`""`**, which are often crucial in user inputs, configurations, or database values.

---
&nbsp;

### 1. Default Configuration Values
```javascript
function createUserConfig(config) {
  // Equivalent to: config.theme = config.theme ?? "light";
  config.theme ??= "light";
  config.language ??= "en";
  config.showSidebar ??= true;
  
  return config;
}

console.log(createUserConfig({ language: "fr" }));
/*
{
  language: 'fr',
  theme: 'light',
  showSidebar: true
}
*/
```

### 2. Initializing Object Properties Safely
```javascript
const user = { name: "Rahul" };

// Ensure preferences object always exists before adding properties
user.preferences ??= {};
user.preferences.darkMode ??= false;

console.log(user);
/*
{
  name: 'Rahul',
  preferences: {
    darkMode: false
  }
}
*/
```

### 3. Caching / Lazy Initialization
```javascript
let cache = null;

function expensiveDatabaseCall() {
  console.log("Fetching from DB...");
  return { items: [1, 2, 3] };
}

function getData() {
  // Executes and assigns the result ONLY if cache is null or undefined
  cache ??= expensiveDatabaseCall();
  return cache;
}

console.log(getData()); 
// "Fetching from DB..." (DB call runs)

console.log(getData()); 
// Cached result returned, no DB call made
```

### 4. Form Handling
```javascript
let formData = {
  username: "rahul",
  email: null, // Explicitly empty
};

// Falls back to a default since email is null
formData.email ??= "not-provided@example.com";

console.log(formData);
/*
{
  username: 'rahul',
  email: 'not-provided@example.com'
}
*/
```
